export interface ProbeSubject {
    /** Workspace-relative path; the engine resolves and guards it. */
    path: string;
    /** Absolute path, already checked to be inside the workspace. */
    abs: string;
    /** sha256:<hex> of the file's bytes (for directories: of the sorted listing of child digests). */
    digest: string;
    size: number;
    kind: "file" | "directory";
}
export interface ProbeContext {
    subject: ProbeSubject;
    params: Json;
    /** Read up to `max` bytes from the subject (or a path inside a directory subject). */
    read(relPath?: string, max?: number): Promise<Uint8Array>;
    /** Run an external tool: argv only, no shell, bounded time and output (vendored Orca process-host). */
    exec(program: string, args: string[], opts?: {
        timeoutMs?: number;
        maxOutputBytes?: number;
    }): Promise<{
        code: number | null;
        stdout: string;
        stderr: string;
        timedOut: boolean;
        truncated: boolean;
    }>;
    /** Abort when the case is cancelled. */
    signal: AbortSignal;
}
export interface ProbeOutput {
    result: Json;
    coverage?: Coverage | null;
    limitations?: string[];
    findings?: Finding[];
    /** Not hashed: timings, tool versions. */
    volatile?: Json;
    locator?: {
        offset?: number;
        symbol?: string;
    };
}
export interface ProbeDef {
    id: string;
    version: string;
    /** The predicate recorded in evidence, e.g. "file.identify", "pkg.manifest", "elf.imports". */
    predicate: string;
    summary: string;
    /** Which subjects it applies to, decided from the cheap identification result. */
    appliesTo(ident: {
        format: string;
        kind: "file" | "directory";
    }): boolean;
    /** Platforms where it runs; the reach table records the rest as not covered. */
    platforms: ("linux" | "darwin" | "win32")[];
    /** External programs it needs; missing ones make it unavailable, never silently wrong. */
    requires?: string[];
    cost: "cheap" | "medium" | "heavy";
    run(ctx: ProbeContext): Promise<ProbeOutput>;
}
export interface ProbeEngine {
    list(): ProbeDef[];
    /** Run one probe on a subject; returns the stored evidence record (reuses it when the ProbeKey already exists). */
    run(input: {
        caseId: string | null;
        probeId: string;
        path: string;
        params?: Json;
        producerId?: string;
    }): Promise<ClaimRecord>;
    /** Identify a subject and run every cheap applicable probe (the model-free first pass). */
    sweep(input: {
        caseId: string | null;
        path: string;
        depth?: number;
        budgetMs?: number;
    }): Promise<SweepReport>;
}
export interface SweepReport {
    subject: string;
    evidence: string[];
    findings: (Finding & {
        observation_id: string;
    })[];
    /** Questions the sweep could not close; each becomes an Unknown. */
    questions: string[];
    notExamined: {
        what: string;
        reason: string;
    }[];
    elapsedMs: number;
}
export type Effect = "none" | "read" | "write" | "exec" | "network" | "spawn" | "publish";
export interface ToolContext {
    caseId: string;
    /** "coordinator" or the worker id that is calling. */
    caller: string;
    workspace: string;
    /** Resolves when the person approves; rejects with approval_denied. Policy calls this, tools need not. */
    approve(action: string, detail: Json): Promise<void>;
    signal: AbortSignal;
}
export interface ToolResult {
    /** Short text for the model. Keep under ~8 KB; put bulk behind a handle. */
    text: string;
    /** Machine-readable payload (also returned as MCP structuredContent). */
    data?: Json;
    /** Handle the model can pass to `assay_fetch` for the full payload. */
    handle?: string;
    /** True when the text came from the target (file contents, tool output): it is wrapped as untrusted. */
    untrusted?: boolean;
    isError?: boolean;
}
export interface ToolDef {
    name: string;
    description: string;
    inputSchema: Json;
    effect: Effect;
    /** Only the coordinator may call it (worker management). */
    coordinatorOnly?: boolean;
    handler(ctx: ToolContext, args: Record<string, Json>): Promise<ToolResult>;
}

    #private;
    readonly db: DatabaseSync;
    readonly path: string;

        from: number;
        to: number;
        backup: string | null;
    };

     * Copy the whole store to `dest` page by page (SQLite's online backup), so an export never holds

     */
    exportTo(dest: string): Promise<{
        pages: number;
    }>;

 * Signal the child's whole tree.
 *
 * POSIX precondition: the child must have been spawned `detached`. The signal
 * goes to the process group `-child.pid`, so a child that is not its own group
 * leader would hand it to whatever group it inherited instead.
 *

//// vendor/orca-process-host/windows-command-line.d.ts
/**
 * Windows command-line construction.
 *
 * Two parsers read the line Orca hands to CreateProcess, and they disagree:
 *
 * - The target program's own startup code (`CommandLineToArgvW`, which the CRT and
 *   Node both use) understands `\"` as a literal quote and `\\` as a literal
 *   backslash.
 * - `cmd.exe` — unavoidable for a `.cmd`/`.bat` target — does NOT. It counts `"`
 *   naively to track quote state, and inside that state it still expands `%VAR%`.
 *
 * So a `\"` written for the first parser silently flips cmd's quote parity, and
 * every later `&` `|` `<` `>` on the line stops being data and becomes an
 * operator. Measured on Windows 11 with a real `.cmd` shim: `["a b", 'c"d',
 * "e%F%g", "h&i", "j^k"]` came back as `["a b", 'c"d', "e^%F^%g", "h"]` — the
 * `&` truncated the argument AND ran the remainder as a command.
 *
 * The encoding below satisfies both parsers at once and round-trips all of the
 * adversarial cases; see windows-command-line.test.ts for the corpus.
 */
/** Quote one argument for `CommandLineToArgvW`. */
export declare function quoteWindowsArgument(value: string): string;
/**
 * Quote one argument that has to survive a cmd hop as well.
 *
 * Separate from `quoteWindowsArgument` rather than a boolean parameter: the
 * two differ only in whether `%` is neutralised, and a flag there is easy to
 * pass by accident — `values.map(quoteWindowsArgument)` hands `map`'s index in
 * as the flag, which is exactly how this was first written.
 */
export declare function quoteWindowsCmdArgument(value: string): string;
/**
 * Decode a command line the way `CommandLineToArgvW` does, e.g. one read back from
 * the process table. `""` inside a quoted run yields a literal quote.
 */
export declare function parseWindowsCommandLine(line: string): string[];
/** Thrown before spawn when a `.cmd`/`.bat` target would receive a line break. Typed so
 * callers can tell the user the batch file is the limit, not the agent. */
export declare class WindowsCmdLineBreakError extends Error {
    constructor();
}
export declare function validateWindowsCmdArguments(values: readonly string[]): void;
/**
 * Build the argv Node should spawn to run `program` with `args` through
 * `cmd.exe`, for targets cmd must interpret (`.cmd`, `.bat`).
 *
 * Callers must pass the result with `windowsVerbatimArguments: true` so Node
 * hands the line through untouched, and must NOT set `shell: true` — that
 * concatenates arguments without escaping (Node warns DEP0190) and silently
 * disables `windowsHide`.
 *
 * `/d` skips AutoRun commands from the registry, which would otherwise run
